ros2 run chassis keyboard_teleop --ros-args -p linear_speed:=0.1

# 按键：
# W / ↑：前进
# S / ↓：后退
# A / ←：左移
# D / →：右移
# 空格：停止
# Q：停止并退出


==================== 常用 ROS 2 检查 ====================

ros2 node list
ros2 topic list -t
ros2 topic echo /cmd_vel
ros2 topic echo /cmd_vel_input
ros2 topic echo /heading_controller/angular_z
ros2 topic echo /car/imu_raw

# 查看话题连接关系
ros2 topic info /cmd_vel -v
ros2 topic info /heading_controller/angular_z -v

# 查看 chassis 包提供的可执行程序
ros2 pkg executables chassis


==================== 手动测试平移 ====================

# 不要直接向 /cmd_vel 发布；应向融合节点输入 /cmd_vel_input 发布
ros2 topic pub --rate 10 /cmd_vel_input geometry_msgs/msg/Twist "{linear: {x: 0.1, y: 0.0, z: 0.0}, angular: {x: 0.0, y: 0.0, z: 0.0}}"
